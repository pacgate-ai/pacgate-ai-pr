 310:       # same or the published image silently loses the PacGate customizations.
 311:       # Files are copied (not git-patched) because they contain long Chinese
 312:       # strings that break git-patch parsing - same mechanism as the local script.
 313:       # Verified 2026-09-17: published 0.1.14 had 0 pacgate-marked files in
 314:       # .next vs 9 in the locally-built image - this step closes that gap.
 315:       - name: Apply PacGate frontend source overrides
 316:         run: |
 317:           if [ ! -d deploy/frontend-patches/files ]; then
 318:             echo "ERROR: deploy/frontend-patches/files missing from checkout" && exit 1
 319:           fi
 320:           cp -rv deploy/frontend-patches/files/. deploy/deer-flow-src/frontend/
 321: 
 322:       - name: Build & push deer-flow-frontend-pacgate
 323:         uses: docker/build-push-action@v6
 324:         with:
 325:           context: .
 326:           file: deploy/deer-flow-frontend-pacgate/Dockerfile
 327:           build-args: |
 328:             DEER_FLOW_INTERNAL_GATEWAY_BASE_URL=http://deer-flow:8001
 329:           push: true
 330:           tags: |
 331:             ${{ steps.ns.outputs.prefix }}/deer-flow-frontend-pacgate:${{ steps.tag.outputs.tag }}
 332:             ${{ steps.ns.outputs.prefix }}/deer-flow-frontend-pacgate:latest
 333:           labels: |
 334:             org.opencontainers.image.source=${{ github.server_url }}/${{ github.repository }}
 335:             org.opencontainers.image.revision=${{ github.sha }}
 336:             org.opencontainers.image.version=${{ steps.tag.outputs.tag }}
 337: 
 338:       - name: Verify image manifest (anonymous HEAD)
 339:         run: |
 340:           # Two things this step must get right, both learned the hard way:
 341:           #
 342:           # 1. GHCR returns 404 for a manifest request sent WITHOUT an OCI/Docker
 343:           #    Accept header, even when the package is public. Omitting it made
 344:           #    every run of this workflow fail on a completely successful
 345:           #    release (7/7 runs red before this fix).
 346:           # 2. GHCR can take a few seconds to propagate a freshly-pushed manifest
 347:           #    across its registry nodes, so retry with backoff instead of racing.
 348:           PREFIX="${{ steps.ns.outputs.prefix }}"
 349:           TAG="${{ steps.tag.outputs.tag }}"
 350:           ACCEPT="application/vnd.oci.image.index.v1+json, application/vnd.docker.distribution.manifest.list.v2+json, application/vnd.docker.distribution.manifest.v2+json"
 351: 
 352:           echo "Verifying ghcr.io/${PREFIX#ghcr.io/}/*:$TAG"
 353:           for img in pacgate-api pacgate-mcp ocr-service deer-flow-pacgate deer-flow-frontend-pacgate; do
 354:             ok=0
 355:             for attempt in 1 2 3 4 5; do
 356:               tok=$(curl -s "https://ghcr.io/token?scope=repository:${PREFIX#ghcr.io/}/${img}:pull" | jq -r .token)
 357:               code=$(curl -s -o /dev/null -w "%{http_code}" \
 358:                 -H "Accept: $ACCEPT" \
 359:                 -H "Authorization: Bearer $tok" \
 360:                 "https://ghcr.io/v2/${PREFIX#ghcr.io/}/${img}/manifests/${TAG}")
 361:               echo "attempt $attempt: ${PREFIX#ghcr.io/}/${img}:${TAG} => HTTP $code"
 362:               if [ "$code" = "200" ]; then ok=1; break; fi
 363:               sleep 10
 364:             done
 365:             if [ "$ok" != "1" ]; then
 366:               # 404 = tag absent in this namespace (likely a namespace mismatch
 367:               # between this workflow and the compose pins, or the tag never
 368:               # reached the registry). 401 = the package is still private.
 369:               echo "::error::${img}:${TAG} is not anonymously pullable (last HTTP $code)."
 370:               echo "If 404: check that the namespace above matches the image pins in"
 371:               echo "deploy/client-bundle/compose.prod.yaml."
 372:               echo "If 401: the package exists but is private - flip it to public in the GHCR UI."
 373:               exit 1
 374:             fi
 375:           done
